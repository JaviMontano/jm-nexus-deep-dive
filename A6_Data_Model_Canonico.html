4Security Rules (Firestore)

rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // --- operators: solo el propio operador lee/escribe --- match /operators/{opId} { allow read, update: if request.auth.uid == opId; allow create: if request.auth != null; allow delete: if false; } // --- projects: owner lee/escribe, NUNCA delete --- match /projects/{projId} { allow read, update: if request.auth.uid == resource.data.owner_id; allow create: if request.auth != null; allow delete: if false; // I-6: nunca borrar } // --- artifacts: owner del proyecto padre --- match /artifacts/{artId} { allow read, write: if request.auth.uid == get(/databases/$(database)/documents/projects/$(resource.data.project_id)).data.owner_id; } // --- pending_actions: solo operator_id resuelve (I-2) --- match /pending_actions/{actId} { allow read: if request.auth.uid == resource.data.operator_id; allow update: if request.auth.uid == resource.data.operator_id && resource.data.expires_at > request.time; // I-3 allow create: if request.auth != null; allow delete: if false; } // --- executions: operator lee/escribe --- match /executions/{exId} { allow read, write: if request.auth.uid == resource.data.operator_id; } // --- short_term_memory: operator, TTL enforced server-side --- match /short_term_memory/{memId} { allow read, write: if request.auth.uid == resource.data.operator_id; } // --- semantic_memory: operator --- match /semantic_memory/{memId} { allow read, write: if request.auth.uid == resource.data.operator_id; } // --- skills / workflows: authenticated read, admin write --- match /skills/{skillId} { allow read: if request.auth != null; allow write: if request.auth.token.admin == true; } match /workflows/{wfId} { allow read: if request.auth != null; allow write: if request.auth.token.admin == true; } // --- audit_log: APPEND-ONLY (I-5) --- match /audit_log/{logId} { allow create: if request.auth != null; allow read: if request.auth.token.admin == true; allow update, delete: if false; // immutable } } }
